FROM python:3.12-slim

WORKDIR /workspace

COPY requirements.txt /workspace/requirements.txt

RUN pip install --no-cache-dir -r /workspace/requirements.txt

ENV PYTHONUNBUFFERED=1
ENV PORT=8080
ENV FILE_MGR_BASE=/workspace

EXPOSE 8080

CMD ["gunicorn", "--no-control-socket", "--workers", "1", "--threads", "4", "--timeout", "120", "--bind", "0.0.0.0:8080", "app.app_hub:app"]
